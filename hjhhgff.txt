local MyriadUI = {}

local TweenService = game:GetService("TweenService")
local UserInputService = game:GetService("UserInputService")

local Theme = {
    Background = Color3.fromRGB(24, 24, 27),
    Sidebar    = Color3.fromRGB(20, 20, 23),
    Row        = Color3.fromRGB(34, 34, 38),
    RowHover   = Color3.fromRGB(42, 42, 47),
    Selected   = Color3.fromRGB(10, 110, 255),
    Border     = Color3.fromRGB(46, 46, 50),
    Text       = Color3.fromRGB(240, 240, 242),
    SubText    = Color3.fromRGB(142, 142, 150),
    Font       = Enum.Font.Gotham,
    FontBold   = Enum.Font.GothamBold,
}

local function New(class, props, parent)
    local inst = Instance.new(class)
    if inst:IsA("GuiObject") then inst.BorderSizePixel = 0 end
    if class == "TextButton" then inst.AutoButtonColor = false end
    for k, v in pairs(props or {}) do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end

local function corner(r, parent)
    return New("UICorner", { CornerRadius = UDim.new(0, r) }, parent)
end

local function tween(inst, props, dur)
    local t = TweenService:Create(inst, TweenInfo.new(dur or 0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    t:Play()
    return t
end

local function makeDraggable(handle, target)
    local dragging, startInput, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            startInput = input.Position
            startPos = target.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    handle.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - startInput
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function protectGui(gui)
    if syn and syn.protect_gui then
        syn.protect_gui(gui)
        gui.Parent = game:GetService("CoreGui")
    elseif gethui then
        gui.Parent = gethui()
    else
        gui.Parent = game:GetService("CoreGui")
    end
end

function MyriadUI:CreateWindow(cfg)
    cfg = cfg or {}
    local title = cfg.Title or "Myriad"
    local subtitle = cfg.Subtitle or ""

    local ScreenGui = New("ScreenGui", {
        Name = "MyriadUI", ResetOnSpawn = false, IgnoreGuiInset = true,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling, DisplayOrder = 999,
    })
    protectGui(ScreenGui)

    local Main = New("CanvasGroup", {
        Name = "Main",
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        Size = UDim2.fromOffset(620, 400),
        BackgroundColor3 = Theme.Background,
        ClipsDescendants = true,
        Parent = ScreenGui,
    })
    corner(12, Main)
    New("UIStroke", { Color = Theme.Border, Thickness = 1 }, Main)
    pcall(function()
        New("UIShadow", {
            Transparency = 0.5, BlurRadius = UDim.new(0, 24),
            Offset = UDim2.new(0, 0, 0, 8), ZIndex = -1,
        }, Main)
    end)

    ------------------------------------------------------------------
    -- شريط العنوان: نقاط macOS + سهمين + اسم النافذة + مربع بحث
    ------------------------------------------------------------------
    local TitleBar = New("Frame", {
        Name = "TitleBar", BackgroundColor3 = Theme.Sidebar, Size = UDim2.new(1, 0, 0, 40),
    }, Main)
    New("Frame", {
        BackgroundColor3 = Theme.Border, BorderSizePixel = 0,
        AnchorPoint = Vector2.new(0, 1), Position = UDim2.new(0, 0, 1, 0), Size = UDim2.new(1, 0, 0, 1),
    }, TitleBar)

    local dotColors = { Color3.fromRGB(255, 95, 87), Color3.fromRGB(254, 188, 46), Color3.fromRGB(40, 200, 64) }
    for i, c in ipairs(dotColors) do
        local dot = New("Frame", {
            BackgroundColor3 = c, Size = UDim2.fromOffset(12, 12),
            Position = UDim2.new(0, 14 + (i - 1) * 20, 0.5, -6),
        }, TitleBar)
        corner(6, dot)
        if i == 1 then
            local hit = New("TextButton", { Text = "", AutoButtonColor = false, BackgroundTransparency = 1, Size = UDim2.fromOffset(20, 20), Position = UDim2.fromOffset(-4, -4) }, dot)
            hit.MouseButton1Click:Connect(function() ScreenGui:Destroy() end)
        end
    end

    local navX = 78
    local sidebarToggle = New("TextButton", {
        Text = "≡", Font = Theme.FontBold, TextSize = 15, TextColor3 = Theme.SubText,
        BackgroundTransparency = 1, AutoButtonColor = false,
        Position = UDim2.fromOffset(navX, 0), Size = UDim2.fromOffset(22, 40),
    }, TitleBar)

    local backBtn = New("TextButton", {
        Text = "‹", Font = Theme.FontBold, TextSize = 17, TextColor3 = Theme.SubText,
        BackgroundTransparency = 1, AutoButtonColor = false,
        Position = UDim2.fromOffset(navX + 26, 0), Size = UDim2.fromOffset(20, 40),
    }, TitleBar)
    local fwdBtn = New("TextButton", {
        Text = "›", Font = Theme.FontBold, TextSize = 17, TextColor3 = Color3.fromRGB(70, 70, 76),
        BackgroundTransparency = 1, AutoButtonColor = false,
        Position = UDim2.fromOffset(navX + 46, 0), Size = UDim2.fromOffset(20, 40),
    }, TitleBar)

    local Breadcrumb = New("TextLabel", {
        Text = title, Font = Theme.FontBold, TextSize = 13, TextColor3 = Theme.Text,
        BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Left,
        Position = UDim2.fromOffset(navX + 74, 0), Size = UDim2.new(0, 140, 1, 0),
    }, TitleBar)

    local SearchBox = New("Frame", {
        BackgroundColor3 = Theme.Row, AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -14, 0.5, 0), Size = UDim2.fromOffset(190, 26),
    }, TitleBar)
    corner(7, SearchBox)
    New("UIStroke", { Color = Theme.Border, Thickness = 1 }, SearchBox)
    New("TextBox", {
        Text = "", PlaceholderText = "Search", PlaceholderColor3 = Theme.SubText,
        Font = Theme.Font, TextSize = 12, TextColor3 = Theme.Text,
        BackgroundTransparency = 1, ClearTextOnFocus = false,
        TextXAlignment = Enum.TextXAlignment.Left,
        Position = UDim2.fromOffset(10, 0), Size = UDim2.new(1, -16, 1, 0),
    }, SearchBox)

    makeDraggable(TitleBar, Main)

    ------------------------------------------------------------------
    -- جسم النافذة: قائمة جانبية + منطقة محتوى
    ------------------------------------------------------------------
    local Body = New("Frame", {
        Name = "Body", BackgroundTransparency = 1,
        Position = UDim2.new(0, 0, 0, 40), Size = UDim2.new(1, 0, 1, -40),
    }, Main)

    local Sidebar = New("ScrollingFrame", {
        Name = "Sidebar", BackgroundColor3 = Theme.Sidebar,
        Size = UDim2.new(0, 188, 1, 0),
        CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y,
        ScrollBarThickness = 2, ScrollBarImageColor3 = Theme.Border,
        BorderSizePixel = 0,
    }, Body)
    New("Frame", {
        BackgroundColor3 = Theme.Border, BorderSizePixel = 0,
        AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, 0, 0, 0), Size = UDim2.new(0, 1, 1, 0),
    }, Sidebar)
    New("UIListLayout", { Padding = UDim.new(0, 2), SortOrder = Enum.SortOrder.LayoutOrder }, Sidebar)
    New("UIPadding", {
        PaddingTop = UDim.new(0, 10), PaddingBottom = UDim.new(0, 10),
        PaddingLeft = UDim.new(0, 10), PaddingRight = UDim.new(0, 10),
    }, Sidebar)

    local ContentArea = New("Frame", {
        Name = "Content", BackgroundTransparency = 1,
        Position = UDim2.new(0, 188, 0, 0), Size = UDim2.new(1, -188, 1, 0),
    }, Body)

    sidebarToggle.MouseButton1Click:Connect(function()
        Sidebar.Visible = not Sidebar.Visible
        ContentArea.Position = Sidebar.Visible and UDim2.new(0, 188, 0, 0) or UDim2.new(0, 0, 0, 0)
        ContentArea.Size = Sidebar.Visible and UDim2.new(1, -188, 1, 0) or UDim2.new(1, 0, 1, 0)
    end)

    ------------------------------------------------------------------
    -- عناصر القائمة الجانبية
    ------------------------------------------------------------------
    local selectedRow = nil
    local Window = { OnNavigate = nil, OnConfigure = nil, OnRun = nil, Nav = {} }

    local function addRow(label, indent, isHeader)
        if isHeader then
            local Header = New("TextLabel", {
                Text = label, Font = Theme.FontBold, TextSize = 11, TextColor3 = Theme.SubText,
                BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Left,
                Size = UDim2.new(1, 0, 0, 22), Position = UDim2.fromOffset(4, 0),
            }, Sidebar)
            return Header
        end

        local Row = New("TextButton", {
            Name = label, Text = "", AutoButtonColor = false, BackgroundTransparency = 1,
            Size = UDim2.new(1, 0, 0, 26),
        }, Sidebar)
        corner(6, Row)

        local Dot = New("Frame", {
            BackgroundColor3 = Theme.SubText,
            AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0, 8 + indent, 0.5, 0),
            Size = UDim2.fromOffset(5, 5),
        }, Row)
        corner(3, Dot)

        local Label = New("TextLabel", {
            Text = label, Font = Theme.Font, TextSize = 12, TextColor3 = Theme.Text,
            BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Left,
            Position = UDim2.fromOffset(22 + indent, 0), Size = UDim2.new(1, -(30 + indent), 1, 0),
        }, Row)

        local function setSelected(on)
            tween(Row, { BackgroundColor3 = on and Theme.Selected or Theme.Sidebar, BackgroundTransparency = on and 0 or 1 }, 0.12)
            tween(Label, { TextColor3 = on and Color3.new(1, 1, 1) or Theme.Text }, 0.12)
            tween(Dot, { BackgroundColor3 = on and Color3.new(1, 1, 1) or Theme.SubText }, 0.12)
        end

        Row.MouseEnter:Connect(function()
            if selectedRow ~= Row then tween(Row, { BackgroundColor3 = Theme.RowHover, BackgroundTransparency = 0 }, 0.12) end
        end)
        Row.MouseLeave:Connect(function()
            if selectedRow ~= Row then tween(Row, { BackgroundTransparency = 1 }, 0.12) end
        end)
        Row.MouseButton1Click:Connect(function()
            for _, r in ipairs(Sidebar:GetChildren()) do
                if r:IsA("TextButton") and r ~= Row then
                    tween(r, { BackgroundTransparency = 1 }, 0.12)
                end
            end
            selectedRow = Row
            setSelected(true)
            if Window.OnNavigate then Window.OnNavigate(label) end
        end)

        Window.Nav[label] = Row
        return Row, setSelected
    end

    addRow("Home", 0, false)
    addRow("Tests", 0, true)
    local standardsRow, selectStandards = addRow("Standards", 8, false)
    addRow("Settings", 22, false)
    addRow("Manual", 22, false)
    addRow("Vulnerabilities", 8, false)
    addRow("Settings", 22, false)
    addRow("Settings", 0, true)
    addRow("Window", 8, false)

    selectedRow = standardsRow
    selectStandards(true)

    ------------------------------------------------------------------
    -- منطقة المحتوى: رأس + Scope + Test Options + Output
    ------------------------------------------------------------------
    local ContentIcon = New("Frame", {
        BackgroundColor3 = Theme.Selected, Position = UDim2.fromOffset(24, 20), Size = UDim2.fromOffset(30, 30),
    }, ContentArea)
    corner(8, ContentIcon)

    New("TextLabel", {
        Text = title, Font = Theme.FontBold, TextSize = 16, TextColor3 = Theme.Text,
        BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Left,
        Position = UDim2.fromOffset(64, 18), Size = UDim2.new(1, -80, 0, 20),
    }, ContentArea)
    New("TextLabel", {
        Text = subtitle, Font = Theme.Font, TextSize = 12, TextColor3 = Theme.SubText,
        BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Left,
        Position = UDim2.fromOffset(64, 37), Size = UDim2.new(1, -80, 0, 16),
    }, ContentArea)

    New("Frame", {
        BackgroundColor3 = Theme.Border, BorderSizePixel = 0,
        Position = UDim2.fromOffset(24, 66), Size = UDim2.new(1, -48, 0, 1),
    }, ContentArea)

    local Scroll = New("ScrollingFrame", {
        BackgroundTransparency = 1, BorderSizePixel = 0,
        Position = UDim2.fromOffset(24, 80), Size = UDim2.new(1, -48, 1, -100),
        CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y,
        ScrollBarThickness = 3, ScrollBarImageColor3 = Theme.Border,
    }, ContentArea)
    New("UIListLayout", { Padding = UDim.new(0, 20), SortOrder = Enum.SortOrder.LayoutOrder }, Scroll)

    local function sectionHeader(text, sub)
        local wrap = New("Frame", { BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, sub and 36 or 18) }, Scroll)
        New("TextLabel", {
            Text = text, Font = Theme.FontBold, TextSize = 13, TextColor3 = Theme.Text,
            BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Left,
            Size = UDim2.new(1, 0, 0, 18),
        }, wrap)
        if sub then
            New("TextLabel", {
                Text = sub, Font = Theme.Font, TextSize = 11, TextColor3 = Theme.SubText,
                BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Left,
                Position = UDim2.fromOffset(0, 18), Size = UDim2.new(1, 0, 0, 16),
            }, wrap)
        end
        return wrap
    end

    -- Scope
    sectionHeader("Scope")
    New("TextLabel", {
        Text = cfg.ScopeText or ("Formally named 'Myriad Environment Specification'. It is an adaptation of the original UNC standard specification."),
        Font = Theme.Font, TextSize = 12, TextColor3 = Theme.SubText, TextWrapped = true,
        BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Left,
        Size = UDim2.new(1, 0, 0, 34),
    }, Scroll)

    -- Test Options
    sectionHeader("Test Options", "Configure the tests")
    do
        local Row = New("Frame", { BackgroundColor3 = Theme.Row, Size = UDim2.new(1, 0, 0, 42) }, Scroll)
        corner(8, Row)
        New("UIStroke", { Color = Theme.Border, Thickness = 1 }, Row)
        New("TextLabel", {
            Text = "Options", Font = Theme.Font, TextSize = 13, TextColor3 = Theme.Text,
            BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Left,
            Position = UDim2.fromOffset(14, 0), Size = UDim2.new(1, -100, 1, 0),
        }, Row)
        local ConfigureBtn = New("TextButton", {
            Text = "Configure", Font = Theme.Font, TextSize = 12, TextColor3 = Theme.Text,
            BackgroundColor3 = Theme.RowHover, AutoButtonColor = false,
            AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0),
            Size = UDim2.fromOffset(84, 26),
        }, Row)
        corner(6, ConfigureBtn)
        New("UIStroke", { Color = Theme.Border, Thickness = 1 }, ConfigureBtn)
        ConfigureBtn.MouseEnter:Connect(function() tween(ConfigureBtn, { BackgroundColor3 = Theme.Border }, 0.12) end)
        ConfigureBtn.MouseLeave:Connect(function() tween(ConfigureBtn, { BackgroundColor3 = Theme.RowHover }, 0.12) end)
        ConfigureBtn.MouseButton1Click:Connect(function()
            if Window.OnConfigure then Window.OnConfigure() end
        end)
        Window.ConfigureButton = ConfigureBtn
    end

    -- Output
    sectionHeader("Output", "View the results of the tests")
    do
        local Row = New("Frame", { BackgroundColor3 = Theme.Row, Size = UDim2.new(1, 0, 0, 42) }, Scroll)
        corner(8, Row)
        New("UIStroke", { Color = Theme.Border, Thickness = 1 }, Row)
        New("TextLabel", {
            Text = "Run Test", Font = Theme.Font, TextSize = 13, TextColor3 = Theme.Text,
            BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Left,
            Position = UDim2.fromOffset(14, 0), Size = UDim2.new(1, -220, 1, 0),
        }, Row)
        New("TextLabel", {
            Text = "⌘ Control + Shift + S", Font = Theme.Font, TextSize = 11, TextColor3 = Theme.SubText,
            BackgroundTransparency = 1, TextXAlignment = Enum.TextXAlignment.Right,
            AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -70, 0.5, 0),
            Size = UDim2.fromOffset(150, 20),
        }, Row)
        local RunBtn = New("TextButton", {
            Text = "Run", Font = Theme.FontBold, TextSize = 12, TextColor3 = Color3.new(1, 1, 1),
            BackgroundColor3 = Theme.Selected, AutoButtonColor = false,
            AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0),
            Size = UDim2.fromOffset(54, 26),
        }, Row)
        corner(6, RunBtn)
        RunBtn.MouseEnter:Connect(function() tween(RunBtn, { BackgroundColor3 = Color3.fromRGB(35, 130, 255) }, 0.12) end)
        RunBtn.MouseLeave:Connect(function() tween(RunBtn, { BackgroundColor3 = Theme.Selected }, 0.12) end)
        RunBtn.MouseButton1Click:Connect(function()
            if Window.OnRun then Window.OnRun() end
        end)
        Window.RunButton = RunBtn
    end

    Window.Instance = Main
    Window.ScreenGui = ScreenGui
    function Window:Destroy() ScreenGui:Destroy() end

    return Window
end

return MyriadUI
